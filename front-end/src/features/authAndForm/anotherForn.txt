import { useState } from "react";
import "./LoginForm.css";

import person_icon from "../../assets/person-icon.png"
import email_icon from "../../assets/email-icon.png"
import password_icon from "../../assets/password.png"

function LoginForm({ onSuccess }) {
  const [isSignup, setIsSignup] = useState(false);
  const [error, setError] = useState("");

  const handleSubmit = async (e) => {
    e.preventDefault();
    
    const { username, password, email } = e.target.elements;

    const payload = {
      username: username.value,
      password: password.value,
      ...(isSignup && { email: email.value }),
    };

    const url = isSignup
      ? "http://localhost:8080/api/auth/register"
      : "http://localhost:8080/api/auth/login";

    const response = await fetch(url, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
      },
      credentials: "include",
      body: JSON.stringify(payload),
    });

    if (response.ok) {
      const data = await response.text(); // або .json() якщо потрібно
      onSuccess(payload.username);
    } else {
      const message = await response.text();
      setError(`Помилка: ${message}`);
    }
  };

return (
    <form className="login__form" onSubmit={handleSubmit}>
        <div className="text_underline">
            <h1>{isSignup ? "Sign Up" : "Login"}</h1>

            <div className="underline"></div>
        </div>
    

        <div className="login__inputs">

            
            <div className={`input animated-email ${!isSignup ? "hidden" : ""}`}>
                <img src={email_icon} alt="" />
                <input
                    type="email"
                    name="email"
                    placeholder="Email"
                    required={isSignup}
                    className="login__input"
                />
            </div>
            
            <div className="input">
                <img src={person_icon} alt="" />
                <input
                    type="text"
                    name="username"
                    placeholder="Username"
                    required
                    className="login__input"
                />
            </div>
            <div className="input">
                <img src={password_icon} alt="" />
                <input
                    type="password"
                    name="password"
                    placeholder="Password"
                    required
                    className="login__input"
                />
            </div>
        </div>

        {error && <div className="login__error">{error}</div>}

        <div className="login__switch">
            <button type="submit" onClick={(e) => {
            if (!isSignup) {
                handleSubmit(e);
            } else {
                setIsSignup(false);
            }
            }} className={!isSignup ? "submit_active" : "submit_unactive"}>
            Login</button>

            <button type="submit" onClick={(e) => {
            if (isSignup) {
                handleSubmit(e);
            } else {
                setIsSignup(true);
            }
            }} className={isSignup ? "submit_active" : "submit_unactive"}>
            Sign Up</button>
        </div>

        <button type="submit" className="login__submit_button">
            {isSignup ? "Зареєструватися" : "Увійти"}
        </button>

    </form>
  );
}

export default LoginForm;
